    <#-- STEP 1: Pre-scan all lines and build handling fee map keyed by parent name -->
    <#assign handlingFeeMap = {}>
    <#list record.item as line>
        <#if line.taxline != 'T'>
            <#assign itemName = (line.item!"")>
            <#if itemName?lower_case?ends_with("- handling fees")>
                <#assign parentName = itemName?lower_case?remove_ending("- handling fees")?trim>
                <#if parentName != "">
                    <#assign handlingFeeMap = handlingFeeMap + { parentName: (line.amount!0) }>
                </#if>
            </#if>
        </#if>
    </#list>

    <#-- STEP 2: Process and render parent lines -->
    <#list record.item as line>
        <#if line.taxline != 'T'>

            <#assign itemName = (line.item!"")>

            <#-- Skip handling fee lines, only accumulate their total -->
            <#if itemName?lower_case?ends_with("- handling fees")>
                <#assign totalHandlingFees += (line.amount!0)>

            <#else>

                <#-- Look up handling fee by name match -->
                <#assign handlingFeeAmount = 0>
                <#assign lookupName = itemName?lower_case?trim>
                <#if handlingFeeMap?keys?seq_contains(lookupName)>
                    <#assign handlingFeeAmount = handlingFeeMap[lookupName]>
                </#if>

                <#assign rentalCostStr = (line.custcol_az_rs_rental_cost!"0")?replace("[^0-9.,٫]", "", "r")?replace(",", "", "r")?replace("٫", ".", "r")>

                <#assign faceStr = (line.custcol_az_rs_faces!"0")?replace("[^0-9.,٫]", "", "r")?replace(",", "", "r")?replace("٫", ".", "r")>
                <#assign face = (faceStr?has_content)?then(faceStr?number, 0)>

                <#assign rateStr = (line.rate!"0")?replace("[^0-9.,٫]", "", "r")?replace(",", "", "r")?replace("٫", ".", "r")>
                <#assign rate = (rateStr?has_content)?then(rateStr?number, 0)>

                <#assign discountPct = line.custcol_az_rs_discount!0>
                <#assign discountAmount = rate * discountPct>
                <#assign net = rate - discountAmount>

                <#assign totalFaces += face>
                <#assign totalRentalCost += rentalCost>
                <#assign totalRate += rate>
                <#assign totalDiscountAmount += discountAmount>
                <#assign totalNet += net>
                <#assign totalNetHand += net + handlingFeeAmount>
