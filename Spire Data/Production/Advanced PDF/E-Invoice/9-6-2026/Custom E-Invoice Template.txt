    <#-- Define Arabic digits as a list -->
    <#assign arabicDigits = ["٠", "١", "٢", "٣", "٤", "٥", "٦", "٧", "٨", "٩"] />

    <#-- Function to convert Western numbers to Arabic -->
    <#function convertToArabicNumber westernNumber>
    <#assign numberString = westernNumber?string />
    <#assign arabicNumber = "" />

    <#-- Process each character in the number string -->
    <#list 0..(numberString?length - 1) as i>
    <#assign char = numberString[i]>
    <#-- Check if the character is a digit -->
    <#if char?matches("\\d")>
    <#assign arabicNumber = arabicNumber + arabicDigits[char?number] />
    <#elseif char == ".">
    <#assign arabicNumber = arabicNumber + "٫" />
    <#-- Replace dot with Arabic decimal separator -->
    <#else>
        <#assign arabicNumber = arabicNumber + char />
        </#if>
    </#list>

    <#-- Return the Arabic number -->
    <#return arabicNumber />
    </#function>
